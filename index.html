(() => { 'use strict'; // Elements const app = document.getElementById('novaApp'); const canvas = document.getElementById('coreCanvas'); const ctx = canvas?.getContext('2d', { alpha: true }); const coreButton = document.getElementById('coreButton'); const coreStatus = document.getElementById('coreStatus'); const coreInstruction = document.getElementById('coreInstruction'); const messageInput = document.getElementById('messageInput'); const sendButton = document.getElementById('sendButton'); const voiceButton = document.getElementById('voiceButton'); const connectionDisplay = document.getElementById('connectionDisplay'); const connectionStatus = document.getElementById('connectionStatus'); const drawer = document.getElementById('chatDrawer'); const drawerBackdrop = document.getElementById('drawerBackdrop'); const settingsSheet = document.getElementById('settingsSheet'); const settingsBackdrop = document.getElementById('settingsBackdrop'); const memoryPanel = document.getElementById('memoryPanel'); const voiceAvailable = 'SpeechRecognition' in window || 'webkitSpeechRecognition' in window; let active = false; let currentState = navigator.onLine ?
